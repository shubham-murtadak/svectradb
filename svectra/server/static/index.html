Developer Quickstart

Connect seamlessly to SvectraDB using the native Python SDK in either embedded in-process mode or remote client-server mode.

Python SDK Client

import svectradb

# 1. Connect to SvectraDB Server
client = svectradb.Client(url="http://localhost:8080")

# 2. Get or Create Collection
collection = client.get_or_create_collection(
    name="ai_papers",
    dimension=128,
    metric="cosine",
    index_type="hnsw"
)

# 3. Upsert Vectors & Payloads
collection.upsert(
    ids=["doc1", "doc2"],
    documents=["Attention is all you need", "SvectraDB vector engine"],
    metadatas=[{"category": "ai", "year": 2026}, {"category": "db", "year": 2026}]
)

# 4. Search with Filter
results = collection.query(
    query_text="neural search",
    top_k=5,
    filter={"category": {"$eq": "ai"}}
)

for r in results:
    print(r.score, r.id, r.document)
